# Qatar4u TV — ADB recovery commands for Galaxy Tab S6 (SM-T865)
# Run from the platform-tools folder after: adb devices (the tablet must show "device").
# Read Restore_Instructions.md first. Run one line at a time.

# 1. Samsung's own home screen back (One UI Home was disabled on 26 Oct 2026)
adb shell pm enable com.sec.android.app.launcher
adb shell cmd package set-home-activity com.sec.android.app.launcher/.activities.LauncherActivity
adb shell input keyevent KEYCODE_HOME

# 3. Settings back to Samsung defaults
adb shell settings put global window_animation_scale 1.0
adb shell settings put global transition_animation_scale 1.0
adb shell settings put global animator_duration_scale 1.0
adb shell settings put global stay_on_while_plugged_in 0

# 4. Optional: remove the custom launchers (your own apps, not system apps)
adb uninstall net.qatar4u.tv
adb uninstall com.gathin.tvhome

# 5. Check
adb shell pm list packages -d
adb shell cmd package resolve-activity --brief -a android.intent.action.MAIN -c android.intent.category.HOME
